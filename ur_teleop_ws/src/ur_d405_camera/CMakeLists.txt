cmake_minimum_required(VERSION 3.16)
project(ur_d405_camera)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(REALSENSE2 REQUIRED IMPORTED_TARGET realsense2)
add_executable(d405_capture src/d405_capture.cpp)
target_link_libraries(d405_capture PkgConfig::REALSENSE2)
target_compile_options(d405_capture PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS d405_capture DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS src/d405_camera.py DESTINATION lib/${PROJECT_NAME} RENAME d405_camera)
ament_package()
