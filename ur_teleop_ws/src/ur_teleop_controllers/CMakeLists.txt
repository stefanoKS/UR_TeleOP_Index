cmake_minimum_required(VERSION 3.16)
project(ur_teleop_controllers)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ur_controllers REQUIRED)
find_package(ruckig REQUIRED)
find_package(pluginlib REQUIRED)
find_package(controller_interface REQUIRED)
find_package(realtime_tools REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(${PROJECT_NAME} SHARED src/smooth_scaled_controller.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(${PROJECT_NAME} ruckig::ruckig)
ament_target_dependencies(${PROJECT_NAME} ur_controllers pluginlib controller_interface
  realtime_tools rclcpp_lifecycle trajectory_msgs std_msgs)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(controller_interface controllers.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_smoothing test/test_smoothing.cpp)
  target_include_directories(test_smoothing PRIVATE include)
  target_link_libraries(test_smoothing ruckig::ruckig)
  ament_add_gtest(test_controller test/test_controller.cpp)
  ament_target_dependencies(test_controller ur_controllers pluginlib controller_interface trajectory_msgs)
endif()
ament_package()
