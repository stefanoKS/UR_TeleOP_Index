cmake_minimum_required(VERSION 3.18)
project(ur_vr_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(trajectory_msgs REQUIRED)

set(LIBSURVIVE_ROOT "$ENV{HOME}/Documents/Teleop/ur_teleop/tools/libsurvive" CACHE PATH "Existing libsurvive checkout")
set(LIBSURVIVE_BUILD_DIR "${LIBSURVIVE_ROOT}/bin" CACHE PATH "Existing libsurvive build directory")
set(survive_cache "${LIBSURVIVE_BUILD_DIR}/CMakeCache.txt")
if(NOT EXISTS "${survive_cache}")
  message(FATAL_ERROR "Build libsurvive first; missing ${survive_cache}")
endif()
file(STRINGS "${survive_cache}" precision_setting REGEX "^USE_SINGLE_PRECISION:BOOL=")
if(NOT precision_setting MATCHES "=(ON|OFF|TRUE|FALSE|0|1)$")
  message(FATAL_ERROR "Cannot determine libsurvive floating point ABI from ${survive_cache}")
endif()
find_library(SURVIVE_LIBRARY NAMES survive PATHS "${LIBSURVIVE_BUILD_DIR}" NO_DEFAULT_PATH REQUIRED)
find_path(SURVIVE_INCLUDE_DIR survive_api.h PATHS "${LIBSURVIVE_ROOT}/include/libsurvive" NO_DEFAULT_PATH REQUIRED)
find_path(CNMATRIX_INCLUDE_DIR cnmatrix/cn_matrix.h PATHS
  "${LIBSURVIVE_ROOT}/libs/cnmatrix/include" "${LIBSURVIVE_ROOT}/libs/cnmatrix"
  NO_DEFAULT_PATH REQUIRED)

add_executable(controller_bridge src/controller_bridge.cpp)
target_include_directories(controller_bridge PRIVATE include)
target_include_directories(controller_bridge SYSTEM PRIVATE
  "${SURVIVE_INCLUDE_DIR}" "${LIBSURVIVE_ROOT}/redist" "${CNMATRIX_INCLUDE_DIR}")
if(precision_setting MATCHES "=(ON|TRUE|1)$")
  target_compile_definitions(controller_bridge PRIVATE USE_FLOAT CN_USE_FLOAT)
else()
  target_compile_definitions(controller_bridge PRIVATE USE_DOUBLE)
endif()
target_compile_options(controller_bridge PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(controller_bridge rclcpp geometry_msgs sensor_msgs std_msgs tf2_ros)
target_link_libraries(controller_bridge "${SURVIVE_LIBRARY}")
set_target_properties(controller_bridge PROPERTIES
  BUILD_RPATH "${LIBSURVIVE_BUILD_DIR}"
  INSTALL_RPATH "${LIBSURVIVE_BUILD_DIR}")
install(TARGETS controller_bridge DESTINATION lib/${PROJECT_NAME})
add_executable(servo_stamp_relay src/servo_stamp_relay.cpp)
ament_target_dependencies(servo_stamp_relay rclcpp trajectory_msgs)
target_compile_options(servo_stamp_relay PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS servo_stamp_relay DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY rviz DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  enable_testing()
  add_executable(test_pose_guard test/test_pose_guard.cpp)
  target_include_directories(test_pose_guard PRIVATE include)
  add_test(NAME pose_guard COMMAND test_pose_guard)
  add_executable(test_optical_guard test/test_optical_guard.cpp)
  target_include_directories(test_optical_guard PRIVATE include)
  add_test(NAME optical_guard COMMAND test_optical_guard)
endif()
ament_package()
